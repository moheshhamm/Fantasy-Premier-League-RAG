// Rule: detect players incorrectly treated as defenders

MATCH (p:Player)-[:PLAYS_AS]->(pos:Position)
WITH p, collect(DISTINCT pos.name) AS positions
WHERE size(positions) = 2 AND "DEF" IN positions

MATCH (p)-[r:PLAYED_IN]->(f:Fixture)
WITH p, r, f,
     // general points
     CASE
       WHEN r.minutes IS NULL OR r.minutes = 0 THEN 0
       WHEN r.minutes < 60 THEN 1
       ELSE 2
     END AS minutes_points,
     coalesce(r.assists, 0) * 3 AS assist_points,
     coalesce(r.penalties_missed, 0) * -2 AS pen_miss_points,
     coalesce(r.own_goals, 0) * -2 AS own_goal_points,
     coalesce(r.yellow_cards, 0) * -1 AS yc_points,
     coalesce(r.red_cards, 0) * -3 AS rc_points,
     // defender-specific points
     coalesce(r.goals_scored, 0) * 6 AS def_goal_points,
     CASE
       WHEN r.minutes >= 60 AND coalesce(r.clean_sheets, 0) > 0 THEN 4
       ELSE 0
     END AS def_cs_points,
     toInteger(coalesce(r.goals_conceded, 0) / 2) * -1 AS def_gc_points

WITH p, f, r,
     minutes_points +
     assist_points +
     pen_miss_points +
     own_goal_points +
     yc_points +
     rc_points +
     def_goal_points +
     def_cs_points +
     def_gc_points AS def_score

WITH p, f, r, def_score,
     (def_score = r.total_points) AS matches_def

WITH p,
     collect({
       def_score: def_score,
       season: f.season,
       fixture_number: f.fixture_number,
       matches_def: matches_def,
       total_points: r.total_points
     }) AS all_fixtures,
     count(*) AS total_fixtures

WITH p, total_fixtures,
     [fx IN all_fixtures WHERE fx.matches_def = false] AS wrong_fixtures
WHERE size(wrong_fixtures) > 0

RETURN
  p.player_name AS player,
  size(wrong_fixtures) AS not_def_fixtures,
  total_fixtures,
  wrong_fixtures AS fixture_details
ORDER BY not_def_fixtures DESC;
